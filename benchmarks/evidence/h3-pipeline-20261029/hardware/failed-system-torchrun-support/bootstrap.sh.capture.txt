#!/usr/bin/env bash
set -euo pipefail
SUPPORT=/workspace/h3-pp-support
OUT=/workspace/h3-pp-output
REPO=/workspace/h3-pp-soup
mkdir -p "$SUPPORT"
exec > >(tee "$SUPPORT/bootstrap.log") 2>&1
finish() {
  code=$?
  printf '%s\n' "$code" > /tmp/h3-pp-bootstrap.exit
  printf '%s\n' "$code" > "$SUPPORT/exit-code.txt"
  cp /tmp/h3-pp-bootstrap.sh "$SUPPORT/bootstrap.sh.capture.txt"
  cp /tmp/h3-pp-gpu-preflight.py "$SUPPORT/gpu-preflight.py.capture.txt"
  cp /tmp/h3-pp-transfer-preflight.json "$SUPPORT/transfer-preflight.json"
  tar -C "$SUPPORT" -czf /workspace/h3-pp-support.tar.gz . || true
  sha256sum /workspace/h3-pp-support.tar.gz > /workspace/h3-pp-support.sha256
  if test "$code" != 0 && test -d "$OUT"; then
    tar --exclude='*.tar.gz*' -C "$OUT" -czf /workspace/h3-pp-diagnostics.tar.gz . || true
    sha256sum /workspace/h3-pp-diagnostics.tar.gz > /workspace/h3-pp-diagnostics.sha256
  fi
}
trap finish EXIT
set -x
git clone --depth=1 --branch koko/h3-fsdp-lora-20261026 \
  https://github.com/SoupTeam/soup-rnd.git "$REPO"
cd "$REPO"
test "$(git rev-parse HEAD)" = e74469aa129b8604da77a6b9b4c8240dffc23ec0
test -z "$(git status --porcelain)"
lscpu > "$SUPPORT/cpu.txt"
free -b > "$SUPPORT/host-memory.txt"
df -h > "$SUPPORT/disk.txt"
test "$(df -Pk /workspace | awk 'NR==2 {print $4}')" -gt 10485760
/usr/bin/python3 -c 'import sys,importlib.metadata as m; assert (3,10)<=sys.version_info[:2]<(3,13); assert m.version("torch").startswith("2.13.0")'
/usr/bin/python3 -m pip uninstall --break-system-packages -y spin
if ! /usr/bin/python3 -m venv --system-site-packages /workspace/h3-pp-venv; then
  apt-get update
  apt-get install -y python3-venv
  /usr/bin/python3 -m venv --system-site-packages /workspace/h3-pp-venv
fi
source /workspace/h3-pp-venv/bin/activate
python -m pip install --no-compile -c benchmarks/h3-fsdp-constraints.txt \
  transformers==5.17.0 peft==0.20.0 accelerate==1.14.0 rich pydantic psutil pyyaml
python -m pip check
timeout 120 torchrun --standalone --nproc-per-node=2 /tmp/h3-pp-gpu-preflight.py
export TOKENIZERS_PARALLELISM=false
timeout 2400 bash benchmarks/h3-pipeline-linux.sh "$OUT"

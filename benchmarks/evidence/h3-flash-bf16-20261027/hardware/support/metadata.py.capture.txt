import hashlib
import json
import os
import sys
import tarfile
from pathlib import Path
root = Path(sys.argv[1])
status = int(sys.argv[2])
campaign = sys.argv[3] if len(sys.argv)>3 else 'campaign'
assert campaign in ('campaign','campaign-deterministic')
files = [p for folder in (root/'live',root/'live'/campaign) if folder.is_dir()
         for p in folder.iterdir() if p.is_file() and p.suffix in ('.json','.txt','.log','.md')]
files += [p for p in (root/'prepared').rglob('*') if p.is_file()]
for path in (root/'live'/campaign).rglob('*'):
    if not path.is_file() or path.name == 'tokenizer.json':
        continue
    if path.suffix in ('.json','.txt','.log','.md'):
        files.append(path)
    elif path.suffix in ('.pt','.safetensors'):
        arm = path.relative_to(root/'live'/campaign).parts[0]
        if not (root/'live'/f'{campaign}-{arm}.tar.json').exists():
            files.append(path)
archive = root/'live'/f'{campaign}-metadata.tar.gz'
manifest = {}
with tarfile.open(str(archive)+'.partial','w:gz',compresslevel=1) as output:
    for path in sorted(set(files)):
        assert not path.is_symlink()
        assert os.path.commonpath([os.path.realpath(root),os.path.realpath(path)]) == os.path.realpath(root)
        name = path.relative_to(root).as_posix()
        manifest[name] = {'bytes':path.stat().st_size,'sha256':hashlib.sha256(path.read_bytes()).hexdigest()}
        output.add(path,arcname=name,recursive=False)
os.replace(str(archive)+'.partial',archive)
value = {'status':'sealed','archive':archive.name,'bytes':archive.stat().st_size,
         'sha256':hashlib.sha256(archive.read_bytes()).hexdigest(),'files':manifest,'experiment_exit':status}
marker = root/'live'/f'{campaign}-metadata.json'
Path(str(marker)+'.partial').write_text(json.dumps(value,indent=2))
os.replace(str(marker)+'.partial',marker)

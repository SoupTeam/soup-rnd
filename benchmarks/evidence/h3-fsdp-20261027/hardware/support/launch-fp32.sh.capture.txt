#!/usr/bin/env bash
set -euo pipefail
cd /workspace/h3-soup
test "$(git rev-parse HEAD)" = "$1"
test -z "$(git status --porcelain)"
test ! -e /tmp/h3-fp32-launched
touch /tmp/h3-fp32-launched
source benchmarks/evidence/h3-fsdp-20261027/venv/bin/activate
export PYTHONPATH="$PWD/src"
export OMP_NUM_THREADS=4
set +e
python benchmarks/harness/h3_fsdp_campaign.py \
  --prepared benchmarks/evidence/h3-fsdp-20261027/prepared \
  --output benchmarks/evidence/h3-fsdp-20261027/live/campaign-fp32 \
  --repeats 2 --measured-steps 10 --timeout-seconds 600 --campaign-seconds 2400 \
  --precision fp32
status=$?
printf '%s\n' "$status" > /tmp/h3-fp32.exit
exit "$status"

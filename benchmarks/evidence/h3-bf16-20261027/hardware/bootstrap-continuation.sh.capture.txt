#!/usr/bin/env bash
set -euo pipefail
H3_REPO=/workspace/h3-soup
H3_ROOT="$H3_REPO/benchmarks/evidence/h3-bf16-replay-20261027"
finish() {
  H3_STATUS=$?
  printf '%s\n' "$H3_STATUS" > /tmp/h3-continuation.exit
  printf '%s\n' "$H3_STATUS" > "$H3_ROOT/live/exit-code.txt"
  /usr/bin/python3 /tmp/h3-metadata.py "$H3_ROOT" "$H3_STATUS"
}
trap finish EXIT
cd "$H3_REPO"
exec > >(tee "$H3_ROOT/live/bootstrap-continuation.log") 2>&1
cp "$0" "$H3_ROOT/live/bootstrap-continuation.sh.capture.txt"
cp "$H3_ROOT/live/final-metadata.json" "$H3_ROOT/live/preparation-failure.json"
/usr/bin/python3 -m pip uninstall --break-system-packages -y spin
source "$H3_ROOT/venv/bin/activate"
python -m pip check
python -m pip freeze > "$H3_ROOT/live/bootstrap-pip-freeze.txt"
export PYTHONPATH="$H3_REPO/src"
export OMP_NUM_THREADS=4
export TOKENIZERS_PARALLELISM=false
python benchmarks/harness/h3_fsdp_run.py prepare --prepared "$H3_ROOT/prepared" \
  --revision 7ae557604adf67be50417f59c2c2f167def9a775
test -f /tmp/h3-transfer-preflight.ok
cp /tmp/h3-transfer-preflight.json "$H3_ROOT/live/transfer-preflight.json"
python benchmarks/harness/h3_fsdp_campaign.py \
  --prepared "$H3_ROOT/prepared" --output "$H3_ROOT/live/campaign" \
  --precision bf16 --bf16-conservative --archive-arms --archive-receipt-timeout 180 \
  --repeats 2 --measured-steps 10 --timeout-seconds 600 --campaign-seconds 1800

import hashlib, json, os, pathlib, subprocess, sys, time
sys.path.insert(0, '/workspace/c2-final/benchmarks/harness')
from arm_validity import ArmWatch
from c2_linux_depth_probe import counters, quiet, memory
from c2_depth_probe import save
p=pathlib.Path
os.chdir('/workspace/c2-final')
base=json.loads(p('/workspace/c2-primitive-1.json').read_text())
assert base['physical_device']=='nvme0n1' and base['correctness_ok'] and base['direct_io_verified']
assert list(counters())==['nvme0n1']
record={'uid':os.getuid(),'git_sha':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'tests_finished_before_check':time.time(),'records':[]}
def watch(name):
 return ArmWatch(arm=name,round=0,run=0,label='C2 manual sensor admission only',required=['suspend','power','foreign_reads'],model_path='/workspace/c2-fixture.bin',disk_device=lambda _:'nvme0n1',disk_reads=lambda device:counters()[device]['read'],other_disk_reads=lambda device:sum(row['read'] for key,row in counters().items() if key!=device))
try:
 record['quiet_before']=quiet()
 with watch('clean idle') as w:
  time.sleep(2)
 record['records'].append(w.record)
 command=[sys.executable,'benchmarks/harness/c2_linux_depth_probe.py','--fixture','/workspace/c2-fixture.bin','--admit-only','--out','/workspace/c2-admission-final.json']
 with watch('waited-for direct-I/O child') as w:
  with p('/workspace/c2-admission-final.log').open('w') as log:
   result=subprocess.run(command,stdout=log,stderr=subprocess.STDOUT,env=dict(os.environ,OMP_NUM_THREADS='8',TOKENIZERS_PARALLELISM='false'),timeout=300)
 record['records'].append(w.record)
 record['child_command']=command
 record['child_exit']=result.returncode
 record['ok']=result.returncode==0 and all(row['outcome']=='ok' for row in record['records'])
 record['physical_memory']=memory()
 record['assets']={}
 for folder in ['/workspace/c2-weights','/workspace/c2-final/c2-cache']:
  files={str(path):{'size':path.stat().st_size,'allocated':path.stat().st_blocks*512,'mtime_ns':path.stat().st_mtime_ns} for path in sorted(p(folder).glob('*')) if path.is_file()}
  record['assets'][folder]={'files':files,'total_bytes':sum(row['size'] for row in files.values()),'allocated_bytes':sum(row['allocated'] for row in files.values())}
except BaseException as error:
 record['error']=repr(error)
 record['ok']=False
finally:
 save(p('/workspace/c2-sensor-manual.json'),record)
